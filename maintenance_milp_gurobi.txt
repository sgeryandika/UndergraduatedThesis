# maintenance_milp_pyomo_rolling.py
# Pyomo MILP for maintenance task packaging with rolling (resettable) intervals.
# - Reads MPD from CSV
# - Selects intervals A and C
# - Enforces bin calendars and activation
# - Assigns tasks to bins or "Line"
# - Rolling logic: each execution resets the next due window
# - Early penalty as linear slack
# - Optional workload balancing
#
# Requires: pandas, pyomo
# Solves with: Gurobi (only invoked in main via SolverFactory)

import math
import pandas as pd
from typing import Dict, Any, List
from pyomo.environ import (
    ConcreteModel, Set, Param, Var, Binary, NonNegativeReals, Reals,
    Constraint, Objective, minimize, value, RangeSet
)
from pyomo.opt import SolverFactory


# ---------- Utilities ----------

def is_multiple(val: float, base: float, tol: float = 1e-9) -> bool:
    if base <= 0:
        return False
    q = val / base
    return abs(q - round(q)) <= tol


def generate_candidates(start: int, end: int, step: int) -> List[int]:
    return [a for a in range(start, end + 1, step)]


# ---------- Data preparation ----------

def prepare_data(
    mpd_csv_path: str,
    interval_A_range=(200, 600, 50),
    interval_C_range=(4000, 9000, 1000),
    planning_horizon: float = 10000.0,
    delta: float = 50.0,
) -> Dict[str, Any]:
    # Read MPD CSV: required columns
    mpd = pd.read_csv(mpd_csv_path)
    required_cols = ["task_description", "task_interval", "task_manhours", "task_zone"]
    for c in required_cols:
        if c not in mpd.columns:
            raise ValueError(f"Missing column '{c}' in MPD CSV.")

    I = list(mpd.index)  # task ids = DataFrame index
    h = {i: float(mpd.loc[i, "task_interval"]) for i in I}     # intervals
    mh = {i: float(mpd.loc[i, "task_manhours"]) for i in I}    # manhours
    zone = {i: str(mpd.loc[i, "task_zone"]) for i in I}        # zones
    tname = {i: str(mpd.loc[i, "task_description"]) for i in I}

    # Horizon and discrete time grid
    H = float(planning_horizon)
    Delta = float(delta)
    T = int(math.ceil(H / Delta))
    Tset = list(range(1, T + 1))
    t_hours = {t: t * Delta for t in Tset}  # map index -> actual hour

    # Interval candidates and their multiples on the grid
    a_start, a_end, a_step = interval_A_range
    c_start, c_end, c_step = interval_C_range
    A_candidates = generate_candidates(a_start, a_end, a_step)
    C_candidates = generate_candidates(c_start, c_end, c_step)

    T_a = {a: [t for t in Tset if is_multiple(t_hours[t], a)] for a in A_candidates}
    T_c = {c: [t for t in Tset if is_multiple(t_hours[t], c)] for c in C_candidates}

    # Occurrence count if task goes to Line
    occ = {i: int(math.ceil(H / h[i])) if h[i] > 0 else 0 for i in I}

    # Rolling logic requires precomputed predecessor indices for max-gap constraints:
    # pred[i][s] = largest t s.t. t_hours[t] <= t_hours[s] - h[i], else 0 if none.
    pred = {i: {} for i in I}
    for i in I:
        for s in Tset:
            cutoff = t_hours[s] - h[i]
            if cutoff <= 0:
                pred[i][s] = 0
            else:
                # linear scan is fine for moderate T; could be optimized with binary search
                pred_idx = max([t for t in Tset if t_hours[t] <= cutoff], default=0)
                pred[i][s] = pred_idx

    # Upper bound on number of executions per task: ceil(H / h_i) + 1 (buffer)
    Kmax = {i: int(math.ceil(H / h[i])) + 1 for i in I}

    # M_i: tight cap on number of potential scheduled executions equals Kmax[i]
    M_i = {i: Kmax[i] for i in I}

    # Flags for interval logic:
    # gA[i,a] = 1 if h_i is multiple of a; gC[i,c] likewise
    gA = {(i, a): 1 if is_multiple(h[i], a) else 0 for i in I for a in A_candidates}
    gC = {(i, c): 1 if is_multiple(h[i], c) else 0 for i in I for c in C_candidates}
    # q[i,a] = 1 if h_i >= a else 0 (used to force Line when chosen A > h_i)
    q = {(i, a): 1 if h[i] >= a - 1e-9 else 0 for i in I for a in A_candidates}

    return {
        "mpd": mpd,
        "I": I,
        "h": h,
        "mh": mh,
        "zone": zone,
        "tname": tname,
        "H": H,
        "Delta": Delta,
        "Tset": Tset,
        "t_hours": t_hours,
        "A_candidates": A_candidates,
        "C_candidates": C_candidates,
        "T_a": T_a,
        "T_c": T_c,
        "occ": occ,
        "pred": pred,
        "Kmax": Kmax,
        "M_i": M_i,
        "gA": gA,
        "gC": gC,
        "q": q,
    }


# ---------- Model build ----------

def build_pyomo_model(
    data: Dict[str, Any],
    # Business parameters
    labor_rate=100.0,
    labor_uplift=4/3,
    revenue_per_hour=500.0,
    overhead_A=1000.0,
    overhead_C=3000.0,
    overhead_Line=2000.0,
    downtime_per_visit_A=2.0,
    downtime_per_visit_C=5.0,
    downtime_per_Line=3.0,
    # Knobs
    workload_balance_weight=0.0,
) -> ConcreteModel:

    m = ConcreteModel()

    # Sets
    I = data["I"]
    Tset = data["Tset"]
    A_candidates = data["A_candidates"]
    C_candidates = data["C_candidates"]
    m.I = Set(initialize=I, ordered=True)
    m.T = Set(initialize=Tset, ordered=True)
    m.ACAND = Set(initialize=A_candidates, ordered=True)
    m.CCAND = Set(initialize=C_candidates, ordered=True)

    # Parameters
    m.h = Param(m.I, initialize=data["h"])
    m.mh = Param(m.I, initialize=data["mh"])
    m.occ = Param(m.I, initialize=data["occ"])
    m.Mi = Param(m.I, initialize=data["M_i"])
    m.Delta = Param(initialize=float(data["Delta"]))

    # Economic parameters
    m.phi = Param(initialize=float(labor_uplift))
    m.rate = Param(initialize=float(labor_rate))
    m.rev = Param(initialize=float(revenue_per_hour))
    m.ohA = Param(initialize=float(overhead_A))
    m.ohC = Param(initialize=float(overhead_C))
    m.ohL = Param(initialize=float(overhead_Line))
    m.tauA = Param(initialize=float(downtime_per_visit_A))
    m.tauC = Param(initialize=float(downtime_per_visit_C))
    m.tauL = Param(initialize=float(downtime_per_Line))

    # Helper data
    T_a = data["T_a"]
    T_c = data["T_c"]
    t_hours = data["t_hours"]
    pred = data["pred"]
    Kmax = data["Kmax"]
    q = data["q"]

    # For each t, which candidates include t as a multiple?
    A_covers_t = {t: [a for a in A_candidates if t in T_a[a]] for t in Tset}
    C_covers_t = {t: [c for c in C_candidates if t in T_c[c]] for t in Tset}

    # Decision variables
    # Interval selection
    m.sA = Var(m.ACAND, domain=Binary)  # pick exactly one A
    m.sC = Var(m.CCAND, domain=Binary)  # pick exactly one C

    # Structural availability at time t
    m.zAhat = Var(m.T, domain=Binary)
    m.zChat = Var(m.T, domain=Binary)

    # Conducted visits at time t, and bin activation
    m.vA = Var(m.T, domain=Binary)
    m.vC = Var(m.T, domain=Binary)
    m.v = Var(m.T, domain=Binary)

    # Task-to-bin assignment and Line decision
    m.x = Var(m.I, m.T, domain=Binary)  # aggregate assignment (sum over k-th executions)
    m.L = Var(m.I, domain=Binary)       # task assigned to Line category

    # A-only linearization for overhead (1 if A-only at t)
    m.uAonly = Var(m.T, bounds=(0.0, 1.0))

    # Peak workload (optional)
    m.Wpeak = Var(domain=NonNegativeReals)

    # Rolling executions:
    # For each task i, k in {1..Kmax[i]}, t in T: xk[i,k,t] = 1 if k-th execution happens at t
    # Existence flag uk[i,k] = 1 if the k-th execution exists
    xk_index = [(i, k, t) for i in I for k in range(1, Kmax[i] + 1) for t in Tset]
    uk_index = [(i, k) for i in I for k in range(1, Kmax[i] + 1)]
    m.xk = Var(xk_index, domain=Binary)
    m.uk = Var(uk_index, domain=Binary)

    # Continuous times of occurrences (constructed as convex combinations)
    m.curr_time = Var(uk_index, domain=Reals)
    m.prev_time = Var(uk_index, domain=Reals)

    # Early slack e[i,k] >= h_i + prev_time - curr_time, 0 if execution doesn’t exist
    m.e = Var(uk_index, domain=NonNegativeReals)

    # ---------- Constraints ----------

    # Choose exactly one interval A and C
    m.choose_one_A = Constraint(expr=sum(m.sA[a] for a in m.ACAND) == 1)
    m.choose_one_C = Constraint(expr=sum(m.sC[c] for c in m.CCAND) == 1)

    # Structural availability equals chosen-interval multiples pattern
    def _A_mult_rule(m, t):
        return m.zAhat[t] == sum(m.sA[a] for a in A_covers_t[t])
    m.A_mult = Constraint(m.T, rule=_A_mult_rule)

    def _C_mult_rule(m, t):
        return m.zChat[t] == sum(m.sC[c] for c in C_covers_t[t])
    m.C_mult = Constraint(m.T, rule=_C_mult_rule)

    # Conduct visits only where structurally available
    m.vA_le_hat = Constraint(m.T, rule=lambda m, t: m.vA[t] <= m.zAhat[t])
    m.vC_le_hat = Constraint(m.T, rule=lambda m, t: m.vC[t] <= m.zChat[t])

    # Bin activation and no “ghost bins”
    m.v_ge_vA = Constraint(m.T, rule=lambda m, t: m.v[t] >= m.vA[t])
    m.v_ge_vC = Constraint(m.T, rule=lambda m, t: m.v[t] >= m.vC[t])
    m.v_le_sum = Constraint(m.T, rule=lambda m, t: m.v[t] <= m.vA[t] + m.vC[t])

    # Line rule: if chosen A > h_i then L_i = 1
    def _line_rule(m, i):
        # sum_a sA[a] * (1 - q[i,a]) equals 1 if chosen A > h_i
        return m.L[i] >= sum(m.sA[a] * (1 - q[(i, a)]) for a in m.ACAND)
    m.line_rule = Constraint(m.I, rule=_line_rule)

    # Assignment only in active bins; cap total scheduled occurrences by Kmax via Mi
    def _x_le_v(m, i, t):
        return m.x[i, t] <= m.v[t]
    m.x_le_v = Constraint(m.I, m.T, rule=_x_le_v)

    def _x_count_cap(m, i):
        return sum(m.x[i, t] for t in m.T) <= (1 - m.L[i]) * m.Mi[i]
    m.x_count_cap = Constraint(m.I, rule=_x_count_cap)

    # Link aggregate assignment to k-indexed occurrences: x[i,t] = sum_k xk[i,k,t]
    def _x_link(m, i, t):
        return m.x[i, t] == sum(m.xk[i, k, t] for k in range(1, Kmax[i] + 1))
    m.x_link = Constraint(m.I, m.T, rule=_x_link)

    # Define existence: each k either occurs once (uk=1) or not at all (uk=0)
    def _uk_def(m, i, k):
        return sum(m.xk[i, k, t] for t in m.T) == m.uk[i, k]
    m.uk_def = Constraint(uk_index, rule=_uk_def)

    # First execution must occur in [0, h_i] if not Line (equals 1 - L[i])
    T_first = {i: [t for t in Tset if t_hours[t] <= data["h"][i] + 1e-9] for i in I}
    def _first_exec(m, i):
        return sum(m.xk[i, 1, t] for t in T_first[i]) == (1 - m.L[i])
    m.first_exec = Constraint(m.I, rule=_first_exec)

    # Forbid first execution outside its initial window
    def _first_forbid_filter(i, t):
        return t not in T_first[i]
    m.first_forbid = Constraint(
        [(i, t) for i in I for t in Tset if _first_forbid_filter(i, t)],
        rule=lambda m, i, t: m.xk[i, 1, t] == 0
    )

    # Existence must be nonincreasing across k: uk[k+1] ≤ uk[k]
    def _exist_chain(m, i, k):
        if k == Kmax[i]:
            return Constraint.Skip
        return m.uk[i, k + 1] <= m.uk[i, k]
    m.exist_chain = Constraint([(i, k) for i in I for k in range(1, Kmax[i])], rule=_exist_chain)

    # Nondecreasing time: cumulative form ensures k+1-th not before k-th
    def _nondecreasing(m, i, k, s):
        if k == Kmax[i]:
            return Constraint.Skip
        return sum(m.xk[i, k + 1, t] for t in m.T if t <= s) <= sum(m.xk[i, k, t] for t in m.T if t <= s)
    m.nondecreasing = Constraint(
        [(i, k, s) for i in I for k in range(1, Kmax[i]) for s in Tset],
        rule=_nondecreasing
    )

    # Max-gap ≤ h_i via predecessor index: by time s, count of (k+1) ≥ count of k up to s - h_i
    def _max_gap(m, i, k, s):
        if k == Kmax[i]:
            return Constraint.Skip
        s_pred = pred[i][s]
        if s_pred == 0:
            return sum(m.xk[i, k + 1, t] for t in m.T if t <= s) >= 0
        return sum(m.xk[i, k + 1, t] for t in m.T if t <= s) >= sum(m.xk[i, k, t] for t in m.T if t <= s_pred)
    m.max_gap = Constraint(
        [(i, k, s) for i in I for k in range(1, Kmax[i]) for s in Tset],
        rule=_max_gap
    )

    # Construct occurrence times from selections
    def _curr_time_def(m, i, k):
        return m.curr_time[i, k] == sum(t_hours[t] * m.xk[i, k, t] for t in m.T)
    m.curr_time_def = Constraint(uk_index, rule=_curr_time_def)

    def _prev_time_def(m, i, k):
        if k == 1:
            return m.prev_time[i, k] == 0.0
        return m.prev_time[i, k] == sum(t_hours[t] * m.xk[i, k - 1, t] for t in m.T)
    m.prev_time_def = Constraint(uk_index, rule=_prev_time_def)

    # Early slack: e[i,k] ≥ h_i + prev_time - curr_time, and turn off if uk=0
    bigE = max(t_hours.values()) + max(data["h"].values())
    def _early_slack(m, i, k):
        return m.e[i, k] >= data["h"][i] + m.prev_time[i, k] - m.curr_time[i, k]
    m.early_slack = Constraint(uk_index, rule=_early_slack)

    def _slack_off(m, i, k):
        return m.e[i, k] <= bigE * m.uk[i, k]
    m.slack_off = Constraint(uk_index, rule=_slack_off)

    # Overhead linearization: uAonly = 1 if A-only (vA==1, vC==0) else 0
    m.uAonly_le_vA = Constraint(m.T, rule=lambda m, t: m.uAonly[t] <= m.vA[t])
    m.uAonly_le_notC = Constraint(m.T, rule=lambda m, t: m.uAonly[t] <= 1 - m.vC[t])
    m.uAonly_ge_diff = Constraint(m.T, rule=lambda m, t: m.uAonly[t] >= m.vA[t] - m.vC[t])

    # Peak manhours constraint (optional)
    def _peak_rule(m, t):
        return sum(m.mh[i] * m.x[i, t] for i in m.I) <= m.Wpeak
    m.peak = Constraint(m.T, rule=_peak_rule)

    # ---------- Objective (cost components) ----------

    # 1) Direct labor over all scheduled assignments
    Direct = sum(m.phi * m.rate * m.mh[i] * m.x[i, t] for i in m.I for t in m.T)

    # 2) Downtime revenue loss per visit type
    RevenueLoss = sum(m.tauA * m.rev * m.vA[t] for t in m.T) + sum(m.tauC * m.rev * m.vC[t] for t in m.T)

    # 3) Overhead (C dominates if both; A-only captured by uAonly)
    Overhead = sum(m.ohC * m.vC[t] + m.ohA * m.uAonly[t] for t in m.T)

    # 4) Early penalty: rev * sum e[i,k]
    EarlyPenalty = m.rev * sum(m.e[i, k] for (i, k) in uk_index)

    # 5) Line category cost per task
    LineCost = sum(m.L[i] * m.occ[i] * (m.phi * m.rate * m.mh[i] + m.tauL * m.rev + m.ohL) for i in m.I)

    # 6) Optional workload balancing penalty
    Balance = float(workload_balance_weight) * m.Wpeak

    m.TotalCost = Objective(expr=Direct + RevenueLoss + Overhead + EarlyPenalty + LineCost + Balance, sense=minimize)

    # Attach meta for reporting
    m._meta = {
        "t_hours": t_hours,
        "tname": data["tname"],
        "A_candidates": A_candidates,
        "C_candidates": C_candidates,
        "Kmax": Kmax,
    }
    return m


# ---------- Solve and report ----------

def solve_and_report(
    m: ConcreteModel,
    data: Dict[str, Any],
    solver_name: str = "gurobi",
    time_limit: float = None,
    mip_gap: float = None,
    log_to_console: bool = True,
):
    opt = SolverFactory(solver_name)
    if time_limit is not None:
        opt.options["TimeLimit"] = float(time_limit)
    if mip_gap is not None:
        opt.options["MIPGap"] = float(mip_gap)
    if solver_name == "gurobi" and not log_to_console:
        opt.options["OutputFlag"] = 0

    results = opt.solve(m, tee=log_to_console)

    # Extract chosen intervals
    A_candidates = data["A_candidates"]
    C_candidates = data["C_candidates"]
    chosen_A = [a for a in A_candidates if value(m.sA[a]) > 0.5]
    chosen_C = [c for c in C_candidates if value(m.sC[c]) > 0.5]
    chosen_A = chosen_A[0] if len(chosen_A) == 1 else None
    chosen_C = chosen_C[0] if len(chosen_C) == 1 else None

    # Build assignment matrix for active bins
    t_hours = m._meta["t_hours"]
    active_bins = [t for t in m.T if value(m.v[t]) > 0.5]
    rows = []
    for t in sorted(active_bins, key=lambda x: t_hours[x]):
        btype = "C" if value(m.vC[t]) > 0.5 else ("A" if value(m.vA[t]) > 0.5 else "-")
        row = {"bin_index": int(t), "bin_hour": float(t_hours[t]), "bin_type": btype}
        for i in m.I:
            row[f"task_{int(i)}"] = int(round(value(m.x[i, t])))
        rows.append(row)
    assignment_df = pd.DataFrame(rows)

    # Line tasks
    line_tasks = [int(i) for i in m.I if value(m.L[i]) > 0.5]

    # Cost components
    def val(expr): return float(value(expr))
    direct_val = val(sum(m.phi * m.rate * m.mh[i] * m.x[i, t] for i in m.I for t in m.T))
    revenue_loss_val = val(sum(m.tauA * m.rev * m.vA[t] for t in m.T) + sum(m.tauC * m.rev * m.vC[t] for t in m.T))
    overhead_val = val(sum(m.ohC * m.vC[t] + m.ohA * m.uAonly[t] for t in m.T))
    early_penalty_val = val(m.rev * sum(m.e[i, k] for (i, k) in [(i, k) for i in data["I"] for k in range(1, data["Kmax"][i] + 1)]))
    line_cost_val = val(sum(m.L[i] * m.occ[i] * (m.phi * m.rate * m.mh[i] + m.tauL * m.rev + m.ohL) for i in m.I))
    balance_val = 0.0  # computed implicitly through objective; optionally recompute if weight > 0
    total_obj = val(m.TotalCost)

    # Pretty print
    print("\n=== Solution summary ===")
    print(f"Chosen interval_A: {chosen_A}")
    print(f"Chosen interval_C: {chosen_C}")
    print(f"Active bins: {len(active_bins)} of {len(m.T)}")
    print(f"Line tasks: {len(line_tasks)}")

    tname = m._meta["tname"]
    if not assignment_df.empty:
        table = assignment_df.copy()
        # rename to task names
        for i in m.I:
            col = f"task_{int(i)}"
            newcol = tname[int(i)]
            table[newcol] = table[col]
            table.drop(columns=[col], inplace=True)
        table = table[["bin_index", "bin_hour", "bin_type"] + [tname[int(i)] for i in m.I]]
        print("\nAssignment matrix (rows=bins, columns=tasks):")
        print(table.to_string(index=False))
    else:
        print("\nNo active bins.")

    if line_tasks:
        print("\nTasks assigned to Line:")
        for i in line_tasks:
            print(f" - {tname[i]} (interval={data['h'][i]}, manhours={data['mh'][i]}, zone={data['zone'][i]}, occurrences={data['occ'][i]})")
    else:
        print("\nNo tasks assigned to Line.")

    print("\nCost breakdown:")
    print(f" - Direct labor:          {direct_val:,.2f}")
    print(f" - Downtime revenue loss: {revenue_loss_val:,.2f}")
    print(f" - Activation overhead:   {overhead_val:,.2f}")
    print(f" - Early penalty:         {early_penalty_val:,.2f}")
    print(f" - Line cost:             {line_cost_val:,.2f}")
    print(f" = Total objective:       {total_obj:,.2f}")

    return {
        "chosen_A": chosen_A,
        "chosen_C": chosen_C,
        "assignment_df": assignment_df,
        "line_tasks": line_tasks,
        "costs": {
            "direct_labor": direct_val,
            "downtime_loss": revenue_loss_val,
            "overhead": overhead_val,
            "early_penalty": early_penalty_val,
            "line_cost": line_cost_val,
            "balance": balance_val,
            "total": total_obj,
        },
    }
