"""
mtp_baseline_refactor.py

Refactored baseline program implementing the thesis (block / equalized)
maintenance task packaging logic and cost model.

Designed to be a readable, modular, and drop-in replacement for the
thesis' Program For MTP.py / CSVandDef.py / Output.py (Attachments A).
(Reference: thesis attachments A.1-A.3). 

Features:
- Read MPD CSV (flexible columns)
- Block & Equalized assignment for A- and C-checks
- Support for C-equalized splitting across A-bins
- Optional 1/2 A-check (half-A) and unpacked/individual handling
- Cost computation: direct labour, material, downtime (shifts), overhead, line cost, early penalties
- Grid search mode for A/C/packs combinations and Excel export
"""

# ------------------------------
# Utilities / Data structures
# ------------------------------
@dataclass
class Task:
    task_id: str
    description: str
    interval_fh: float
    manhours: float
    # optionally provided in CSV: 'group' or 'check_type' (e.g. 'A' or 'C')
    check_type: Optional[str] = None
    # any additional fields
    extra: dict = None


@dataclass
class Params:
    # Economic / cost parameters
    phi: float = 4/3                  # labour uplift factor
    labor_rate: float = 100.0         # $ per manhour (labour)
    material_fraction: float = 1/3    # MTC as fraction of labour component
    revenue_per_hour: float = 500.0   # $ revenue per flight hour (for downtime/opportunity cost)
    # Overheads (per activation)
    ohA: float = 1000.0
    ohC: float = 3000.0
    ohL: float = 2000.0
    # Manhours per shift (thesis assump.)
    mh_per_shift_A: float = 48.0
    mh_per_shift_C: float = 90.0
    shift_hours: float = 6.0          # hours per shift
    # line/task settings
    tauL: float = 3.0                 # downtime multiplier or equivalent for Line occurrences (kept for compatibility)
    # horizon & bins
    H: int = 36000
    # output options
    export_excel: bool = False


# ------------------------------
# MPD loader (flexible for MPD CSV formats)
# ------------------------------
def load_mpd_csv(path: str) -> pd.DataFrame:
    """
    Load MPD from CSV into a normalized dataframe.
    Expected columns (some optional):
      - task_id, description, interval_fh, interval_fc, interval_ct, manhours, check_type (A/C)
    The loader will try to parse columns heuristically if names differ.

    The returned dataframe will contain:
      'task_id','description','interval_fh','manhours','check_type'
    """
    df = pd.read_csv(path)
    # Normalize column names lower
    cols = {c.lower(): c for c in df.columns}
    # Find keys heuristically
    col_taskid = cols.get('task_id') or cols.get('id') or cols.get('code') or next((c for c in df.columns if 'id' in c.lower()), df.columns[0])
    col_desc = cols.get('description') or next((c for c in df.columns if 'desc' in c.lower()), col_taskid)
    col_mh = cols.get('manhours') or cols.get('manhour') or next((c for c in df.columns if 'manhour' in c.lower()), None)
    col_ifh = cols.get('interval_fh') or next((c for c in df.columns if 'fh' in c.lower() and 'interval' in c.lower()), None)
    col_ifc = cols.get('interval_fc') or next((c for c in df.columns if 'fc' in c.lower() and 'interval' in c.lower()), None)
    col_ict = cols.get('interval_ct') or next((c for c in df.columns if 'day' in c.lower() or 'calendar' in c.lower()), None)
    col_check = cols.get('check_type') or next((c for c in df.columns if 'check' in c.lower()), None)

    # fill missing mh column if not found
    if col_mh is None:
        raise ValueError("MPD CSV must include manhours column (manhours/manhour).")

    normalized = []
    for _, r in df.iterrows():
        # pick interval: interval_fh if present; else try fc/ct conversions later externally
        interval_fh = None
        if col_ifh and not pd.isna(r[col_ifh]):
            interval_fh = float(r[col_ifh])
        # fallback: if only FC or CT present, set NaN and let caller convert
        manhours = float(r[col_mh]) if not pd.isna(r[col_mh]) else 0.0
        tid = str(r[col_taskid])
        desc = str(r[col_desc]) if col_desc else ''
        check_type = str(r[col_check]) if col_check and not pd.isna(r[col_check]) else None
        normalized.append({
            'task_id': tid,
            'description': desc,
            'interval_fh': interval_fh,
            'manhours': manhours,
            'check_type': check_type
        })
    return pd.DataFrame(normalized)


# ------------------------------
# Conversion helpers
# ------------------------------
def convert_fc_ct_to_fh(df: pd.DataFrame, fc_to_fh_ratio: float = 1.5, days_to_fh_ratio: float = None, daily_utilization: float = None):
    """
    Convert FC/CT columns (if present) to FH using provided ratios/daily utilization.
    If the MPD included only FC or CT intervals, convert them into FH and fill interval_fh.
    - fc_to_fh_ratio: e.g. 1 FC = 1.5 FH default from thesis.
    - days_to_fh_ratio: if you know how many FH per day to convert CT -> FH (or use daily_utilization).
    - daily_utilization: flight hours per day (e.g., 7.5), used to convert CT (days/month) -> FH:
          FH = CT_days * daily_utilization
    This function mutates df in place (expects columns 'interval_fc'/'interval_ct' if present).
    """
    # If df already has numeric 'interval_fh' values, preserve those.
    # This is a convenience filler — actual MPD parsing should supply interval_fh directly ideally.
    if 'interval_fc' in df.columns and df['interval_fc'].notna().any():
        df.loc[df['interval_fh'].isna(), 'interval_fh'] = df.loc[df['interval_fh'].isna(), 'interval_fc'] * fc_to_fh_ratio
    if 'interval_ct' in df.columns and df['interval_ct'].notna().any():
        if daily_utilization is not None:
            df.loc[df['interval_fh'].isna(), 'interval_fh'] = df.loc[df['interval_fh'].isna(), 'interval_ct'] * daily_utilization
        elif days_to_fh_ratio is not None:
            df.loc[df['interval_fh'].isna(), 'interval_fh'] = df.loc[df['interval_fh'].isna(), 'interval_ct'] * days_to_fh_ratio
    return df


# ------------------------------
# Bin generation and helpers
# ------------------------------
def generate_bins(A: int, C: int, half_a: bool = False) -> np.ndarray:
    """
    Return sorted array of bins (flight-hour check times).
    Generates A-bins every A FH up to (but excluding) C, and includes C as a final bin.
    If half_a is True, also include midpoints A/2, 3A/2, ... (for 1/2 A-check scheme).
    """
    if A <= 0 or C <= 0:
        raise ValueError("A and C must be positive")
    # A-bins (multiples of A) up to C (exclusive)
    a_bins = np.arange(A, C, A, dtype=int)
    bins = list(a_bins)
    # optionally add half-A bins (if policy requires)
    if half_a:
        half_step = A // 2
        if half_step > 0:
            half_bins = list(np.arange(half_step, C, half_step, dtype=int))
            # unify
            bins = sorted(set(bins) | set(half_bins))
    # final C bin
    bins = sorted(set(bins) | {int(C)})
    return np.array(bins, dtype=int)


def due_times_for_interval(interval: float, H: int) -> np.ndarray:
    """Return an integer array of due times k*interval up to H."""
    if interval <= 0 or np.isnan(interval):
        return np.array([], dtype=int)
    kmax = int(math.floor(H / interval))
    if kmax <= 0:
        return np.array([], dtype=int)
    dues = np.arange(1, kmax + 1) * int(interval)
    return dues.astype(int)


# ------------------------------
# Assignment: Block & Equalized
# ------------------------------
def assign_block_all(tasks_df: pd.DataFrame, A: int, C: int, H: int, half_a: bool=False) -> pd.DataFrame:
    """
    Block assignment:
      - For each A-bin (multiples of A) up to C and C itself, assign tasks whose interval divides the bin value.
      - Tasks with interval < A are considered Line tasks (not assigned here).
    Returns assignments DataFrame with columns:
      ['task_id','due','bin','mh','method','check_level']
    method: 'A-block' or 'C-block'
    check_level: 'A' or 'C' indicating whether this assignment belongs to A-check group or C-check.
    """
    bins = generate_bins(A, C, half_a=half_a)
    records = []
    for _, row in tasks_df.iterrows():
        tid = row['task_id']
        interval = float(row['interval_fh'])
        mh = float(row['manhours'])
        if math.isnan(interval) or interval <= 0:
            continue
        # If task interval < A => treat as line-task (skip)
        if interval < A:
            continue
        dues = due_times_for_interval(interval, H)
        # For block: a due at d occurs if some bin equals d (i.e., d % interval == 0 and d in bins)
        # Equivalent loop: for each bin b in bins up to H, if b % interval == 0 assign
        for b in bins:
            if b > H: 
                continue
            if (b % interval) == 0:
                # Determine check_level by comparing to C threshold
                check_level = 'C' if b >= C else 'A'
                records.append((tid, int(b), int(b), mh, f'{check_level}-block', check_level))
    if not records:
        return pd.DataFrame(columns=['task_id','due','bin','mh','method','check_level'])
    assignments = pd.DataFrame(records, columns=['task_id','due','bin','mh','method','check_level'])
    return assignments


def assign_equalized_all(tasks_df: pd.DataFrame, A: int, C: int, H: int, half_a: bool=False) -> pd.DataFrame:
    """
    Equalized assignment:
      - For A-level tasks: each due (k*interval) is assigned to the latest A-bin <= due (searchsorted).
      - For C-level tasks: if C-equalized is desired, each C occurrence (k*C) is split across A-bins (C/A parts).
        here we implement C-equalized by splitting each C due into N = C/A portions and assign each portion
        to successive A-bins (the behaviour described in thesis attachments).
    Returns assignments with fractional manhours for split parts when needed.
    """
    bins = generate_bins(A, C, half_a=half_a)
    a_bins = np.array([b for b in bins if b < C], dtype=int)
    records = []
    for _, row in tasks_df.iterrows():
        tid = row['task_id']
        interval = float(row['interval_fh'])
        mh = float(row['manhours'])
        if math.isnan(interval) or interval <= 0:
            continue
        # Line tasks skip
        if interval < A:
            continue
        # If task is A-type (interval < C), use A equalized logic
        if interval < C:
            dues = due_times_for_interval(interval, H)
            if dues.size == 0:
                continue
            # find latest A-bin <= due
            idx = np.searchsorted(a_bins, dues, side='right') - 1
            for d, i in zip(dues, idx):
                if i < 0:
                    # no A-bin before due (rare); skip (could be line)
                    continue
                b = int(a_bins[i])
                slack = float(max(0.0, d - b))
                records.append((tid, int(d), b, mh, 'A-equalized', 'A', slack))
        else:
            # interval >= C -> treat as C-check task
            dues = due_times_for_interval(interval, H)  # these are multiples of C
            if dues.size == 0:
                continue
            # For each C due time, split into N parts across A-bins:
            N = max(1, int(round(C / A)))
            # determine the A-bins that fall strictly within the previous C block:
            # for C due at dC = k*C, the preceding block is from (dC - C) to dC.
            for dC in dues:
                # find A-bins in (dC - C, dC] (there should be N of them if A divides C)
                candidate_abins = a_bins[(a_bins > (dC - C)) & (a_bins <= dC)]
                if candidate_abins.size == 0:
                    # fallback: assign to C bin (dC)
                    records.append((tid, int(dC), int(dC), mh, 'C-equalized-fallback', 'C', 0.0))
                    continue
                # split manhours equally across candidate_abins
                mh_share = float(mh / candidate_abins.size)
                for b in candidate_abins:
                    records.append((tid, int(dC), int(b), mh_share, 'C-equalized', 'C', float(dC - b)))
    if not records:
        return pd.DataFrame(columns=['task_id','due','bin','mh','method','check_level','slack'])
    assignments = pd.DataFrame(records, columns=['task_id','due','bin','mh','method','check_level','slack'])
    return assignments


# ------------------------------
# Cost computations (thesis-style)
# ------------------------------
def compute_bin_downtime_hours(bin_mh: float, is_c_bin: bool, params: Params) -> float:
    """
    Convert total manhours at a bin into downtime hours using per-shift capacity.
    Formula (thesis-inspired):
      shifts_needed = ceil(bin_mh / (mh_per_shift_X))
      downtime_hours = shifts_needed * shift_hours
    Use C shift capacity for C-bins, else A capacity.
    """
    if bin_mh <= 0:
        return 0.0
    if is_c_bin:
        cap = params.mh_per_shift_C
    else:
        cap = params.mh_per_shift_A
    shifts = math.ceil(bin_mh / cap)
    return shifts * params.shift_hours


def compute_costs_from_assignments(assignments: pd.DataFrame, tasks_df: pd.DataFrame, params: Params, A: int, C: int) -> Dict:
    """
    Given an assignments DataFrame (result of assign_block_all or assign_equalized_all),
    compute the cost breakdown:
      - direct labour (phi * labor_rate * sum_mh)
      - material cost (material_fraction * labour_component)
      - overhead per bin (ohA or ohC)
      - downtime cost (downtime_hours * revenue_per_hour)
      - line_cost for tasks with interval < A (estimated separately)
    Returns dict with keys: total_mh, direct, material, overhead, downtime, line_cost, total, avg_cost_per_FH
    """
    # totals
    total_mh = float(assignments['mh'].sum()) if not assignments.empty else 0.0
    direct_labour = params.labor_rate * total_mh
    direct = params.phi * direct_labour
    material = params.material_fraction * direct_labour

    # per-bin aggregates
    bins_used = []
    if not assignments.empty:
        bins_used = sorted(assignments['bin'].unique().tolist())
    overhead_total = 0.0
    downtime_total = 0.0
    for b in bins_used:
        bin_df = assignments[assignments['bin'] == b]
        bin_mh = float(bin_df['mh'].sum())
        is_c_bin = (int(b) == int(C))
        overhead_total += params.ohC if is_c_bin else params.ohA
        downtime_hours = compute_bin_downtime_hours(bin_mh, is_c_bin, params)
        downtime_total += downtime_hours * params.revenue_per_hour

    # line tasks: tasks with original interval < A
    line_cost = 0.0
    # identify tasks in the source tasks_df
    for _, r in tasks_df.iterrows():
        interval = r.get('interval_fh', None)
        if interval is None or math.isnan(interval):
            continue
        if float(interval) < A:
            # occurrences over horizon
            occ = math.ceil(params.H / float(interval))
            mh = float(r['manhours'])
            # per-occurrence direct labour + overhead + downtime (approx)
            occ_lab = params.phi * params.labor_rate * mh
            occ_overhead = params.ohL
            occ_downtime = params.tauL * params.revenue_per_hour
            line_cost += occ * (occ_lab + occ_overhead + occ_downtime)

    # 🔹 Early penalty: always included (Option 1)
    early_cost = 0.0
    if not assignments.empty and 'slack' in assignments.columns:
        early_cost = params.revenue_per_hour * assignments['slack'].sum()

    total = direct + material + overhead_total + downtime_total + line_cost + early_cost
    avg_cost_per_fh = total / params.H if params.H > 0 else float('inf')

    return {
        'total_mh': total_mh,
        'direct': direct,
        'material': material,
        'overhead': overhead_total,
        'downtime': downtime_total,
        'line_cost': line_cost,
        'early_cost': early_cost,
        'total': total,
        'avg_cost_per_fh': avg_cost_per_fh,
    }


# ------------------------------
# Output to Excel (basic, thesis-inspired)
# ------------------------------
def export_results_to_excel(path: str, a_interval: int, c_interval: int, a_packs: int, c_packs: int,
                            assignments: pd.DataFrame, costs: Dict):
    """
    Export a compact report to an Excel workbook. Uses xlsxwriter (thesis used same lib).
    This function creates a simple workbook with summary and assignments table.
    """
    if not _HAS_XLSX:
        print("xlsxwriter not installed; skipping Excel export.")
        return
    wb = xlsxwriter.Workbook(path)
    ws = wb.add_worksheet("Summary")
    row = 0
    ws.write(row, 0, "A-Check Interval")
    ws.write(row, 1, a_interval)
    row += 1
    ws.write(row, 0, "C-Check Interval")
    ws.write(row, 1, c_interval)
    row += 1
    ws.write(row, 0, "A Packs")
    ws.write(row, 1, a_packs)
    row += 1
    ws.write(row, 0, "C Packs")
    ws.write(row, 1, c_packs)
    row += 2
    ws.write(row, 0, "Cost component")
    ws.write(row, 1, "Value")
    row += 1
    for k in ['direct','material','overhead','downtime','line_cost','total']:
        ws.write(row, 0, k)
        ws.write(row, 1, float(costs.get(k, 0.0)))
        row += 1

    # assignments sheet
    ws2 = wb.add_worksheet("Assignments")
    hdr = ['task_id','due','bin','mh','method','check_level','slack']
    for c, h in enumerate(hdr):
        ws2.write(0, c, h)
    for r_idx, r in enumerate(assignments.itertuples(index=False), start=1):
        for c_idx, h in enumerate(hdr):
            ws2.write(r_idx, c_idx, getattr(r, h, None))
    wb.close()
    print(f"Exported results to {path}")


# ------------------------------
# Main high-level driver: run baseline (single config) or loop search
# ------------------------------
def run_baseline(tasks_df: pd.DataFrame, A: int, C: int, H: int, method_A: str='equalized', method_C: str='equalized',
                 half_a: bool=False, params: Params=None, export_excel_path: Optional[str]=None) -> Dict:
    """
    Run baseline assignment & cost for a single configuration.
    method_A/C: 'block' or 'equalized'
    returns dict with assignments DataFrame and cost dict
    """
    p = params if params is not None else Params(H=H)
    # choose assignment engines
    if method_A.lower() == 'block' and method_C.lower() == 'block':
        # block-all mode: use assign_block_all (it does both A & C assignments using bins)
        assignments = assign_block_all(tasks_df, A, C, H, half_a=half_a)
        # block assigns check_level accordingly
    else:
        # combine A-equalized and C-equalized or mixed modes
        # We'll obtain assignments by using equalized engine but filter by interval categories when needed
        # First, build equalized for A-level tasks and C-level tasks
        # For mixed modes we'll modify behaviour:
        if method_A.lower() == 'equalized' and method_C.lower() == 'equalized':
            assignments = assign_equalized_all(tasks_df, A, C, H, half_a=half_a)
        else:
            # Mixed: build separate assignments then union
            recs = []
            # A-part (if method_A == block => emulate block for A-only bins)
            if method_A.lower() == 'block':
                a_assign = assign_block_all(tasks_df[tasks_df['interval_fh'] < C], A, C, H, half_a=half_a)
                if not a_assign.empty:
                    recs.append(a_assign)
            else:
                a_assign = assign_equalized_all(tasks_df[tasks_df['interval_fh'] < C], A, C, H, half_a=half_a)
                if not a_assign.empty:
                    recs.append(a_assign)
            # C-part
            if method_C.lower() == 'block':
                c_assign = assign_block_all(tasks_df[tasks_df['interval_fh'] >= C], A, C, H, half_a=half_a)
                if not c_assign.empty:
                    recs.append(c_assign)
            else:
                c_assign = assign_equalized_all(tasks_df[tasks_df['interval_fh'] >= C], A, C, H, half_a=half_a)
                if not c_assign.empty:
                    recs.append(c_assign)
            assignments = pd.concat(recs, ignore_index=True) if recs else pd.DataFrame(columns=['task_id','due','bin','mh','method','check_level','slack'])
    # compute costs
    costs = compute_costs_from_assignments(assignments, tasks_df, p, A=A, C=C)
    # excel export
    if export_excel_path and p.export_excel:
        export_results_to_excel(export_excel_path, A, C, a_packs=None, c_packs=None, assignments=assignments, costs=costs)
    return {'assignments': assignments, 'costs': costs}


def loop_search_and_report(tasks_df: pd.DataFrame,
                           A_range: List[int],
                           C_range: List[int],
                           a_methods: List[str],
                           c_methods: List[str],
                           params: Params,
                           half_a_options: List[bool] = [False],
                           export_best_excel: Optional[str]=None) -> pd.DataFrame:
    """
    Loop over candidate A and C ranges and packaging methods, compute average cost/FH,
    and return a DataFrame results table. Optionally export best config to Excel.
    """
    rows = []
    best = None
    for A in A_range:
        for C in C_range:
            if C <= A:
                # skip invalid combos
                continue
            for mA in a_methods:
                for mC in c_methods:
                    for half_a in half_a_options:
                        out = run_baseline(tasks_df, A=A, C=C, H=params.H, method_A=mA, method_C=mC, half_a=half_a, params=params)
                        costs = out['costs']
                        rows.append({
                            'A': A, 'C': C, 'method_A': mA, 'method_C': mC, 'half_a': half_a,
                            'avg_cost_per_fh': costs['avg_cost_per_fh'],
                            'total_cost': costs['total'],
                            'total_mh': costs['total_mh']
                        })
                        if best is None or costs['avg_cost_per_fh'] < best['avg_cost_per_fh']:
                            best = {
                                'A': A, 'C': C, 'method_A': mA, 'method_C': mC, 'half_a': half_a,
                                'avg_cost_per_fh': costs['avg_cost_per_fh'],
                                'total_cost': costs['total'],
                                'assignments': out['assignments'],
                                'costs': costs
                            }
    results_df = pd.DataFrame(rows)
    if export_best_excel and best and params.export_excel:
        export_results_to_excel(export_best_excel, best['A'], best['C'], a_packs=None, c_packs=None,
                                assignments=best['assignments'], costs=best['costs'])
    return results_df.sort_values('avg_cost_per_fh'), best
